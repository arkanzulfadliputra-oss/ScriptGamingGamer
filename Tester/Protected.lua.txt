        while walkspeedEnabled do
            task.wait(0.1)

                    humanoid.WalkSpeed = currentSpeed

local instantInteractEnabled = false
local currentHoldDuration = 0.1
local originalHoldDuration = {}

local function saveOriginalValue(prompt)
    if originalHoldDuration[prompt] == nil then
        originalHoldDuration[prompt] = prompt.HoldDuration
    end
end

local function getAllPrompts()
    local prompts = {}
    for _, obj in ipairs(game:GetDescendants()) do
        if obj:IsA("ProximityPrompt") then
            table.insert(prompts, obj)
        end
    end
    return prompts
end

local function startInstantInteract()
    local prompts = getAllPrompts()
    for _, prompt in ipairs(prompts) do
        saveOriginalValue(prompt)
        prompt.HoldDuration = currentHoldDuration
    end
end

local function stopInstantInteract()
    for prompt, originalValue in pairs(originalHoldDuration) do
        if prompt and prompt.Parent then
            prompt.HoldDuration = originalValue
        end
    end
    originalHoldDuration = {}
end

local function onDescendantAdded(instance)
    if instantInteractEnabled and instance:IsA("ProximityPrompt") then
        saveOriginalValue(instance)
        instance.HoldDuration = currentHoldDuration
    end
end

game.DescendantAdded:Connect(onDescendantAdded)

local billboardEnabled = false
local activeBillboards = {}
local billboardLoop = nil

local function createBillboardForTimer(timerTextLabel, screenPart)
    if not screenPart or not screenPart:IsA("BasePart") then
        return nil
    end
    
    local old = screenPart:FindFirstChild("TimerBillboard")
    if old then
        old:Destroy()
    end
    
    local billboard = Instance.new("BillboardGui")
    billboard.Name = "TimerBillboard"
    billboard.Adornee = screenPart
    billboard.Parent = screenPart
    billboard.Size = UDim2.fromOffset(250, 60)
    billboard.StudsOffset = Vector3.new(0, 3, 0)
    billboard.AlwaysOnTop = true
    billboard.MaxDistance = 1000
    
    local textLabel = Instance.new("TextLabel")
    textLabel.Size = UDim2.fromScale(1, 1)
    textLabel.BackgroundTransparency = 1
    textLabel.TextColor3 = Color3.new(1, 1, 1)
    textLabel.TextStrokeTransparency = 0
    textLabel.TextSize = 30
    textLabel.Font = Enum.Font.GothamBold
    textLabel.Parent = billboard
    
    activeBillboards[screenPart] = {
        Billboard = billboard,
        TextLabel = textLabel,
        TimerLabel = timerTextLabel
    }
    
    return billboard
end

local function updateBillboardText()
    for screenPart, data in pairs(activeBillboards) do
        if data.TimerLabel and data.TimerLabel.Parent and data.TextLabel then
            data.TextLabel.Text = " " .. tostring(data.TimerLabel.Text)
        end
    end
end

local function startBillboardLoop()
    if billboardLoop then return end
    
    billboardLoop = task.spawn(function()
        while billboardEnabled do
            updateBillboardText()
            task.wait(0.1)
        end
    end)
end

local function stopBillboardLoop()
    if billboardLoop then
        task.cancel(billboardLoop)
        billboardLoop = nil
    end
end

local function removeAllBillboards()
    for screenPart, data in pairs(activeBillboards) do
        if data.Billboard and data.Billboard.Parent then
            data.Billboard:Destroy()
        end
    end
    activeBillboards = {}
end

local function createAllBillboards()
    removeAllBillboards()
    
    for i = 1, 8 do
        local tycoon = TycoonsFolder:FindFirstChild("Tycoon" .. i)
        if tycoon then
            local timeLabel = tycoon:FindFirstChild("Tycoon")
                and tycoon.Tycoon:FindFirstChild("ForcefieldFolder")
                and tycoon.Tycoon.ForcefieldFolder:FindFirstChild("Screen")
                and tycoon.Tycoon.ForcefieldFolder.Screen:FindFirstChild("Screen")
                and tycoon.Tycoon.ForcefieldFolder.Screen.Screen:FindFirstChild("SurfaceGui")
                and tycoon.Tycoon.ForcefieldFolder.Screen.Screen.SurfaceGui:FindFirstChild("Time")
            
            if timeLabel and timeLabel:IsA("TextLabel") then
                local screenPart = timeLabel.Parent and timeLabel.Parent.Parent
                if screenPart and screenPart:IsA("BasePart") then
                    createBillboardForTimer(timeLabel, screenPart)
                end
            end
        end
    end
end

local function CheckAndLock()
	local myTycoon, tycoonNum = GetOwnerTycoon()

	if not myTycoon then
		return
	end

	local timerText = GetTimerText(myTycoon)

	if not timerText then
		return
	end

	print("Timer:", timerText)

	if timerText == "0s" and not alreadyLocked then
		alreadyLocked = true

		TeleportToButton(myTycoon)

		print("LOCKED! Tycoon " .. tycoonNum)
	end

	if timerText ~= "0s" then
		alreadyLocked = false
	end
end

local function GetDistance(plr)
	local myChar = LocalPlayer.Character
	if not myChar then return 0 end

	local myHRP = myChar:FindFirstChild("HumanoidRootPart")
	if not myHRP then return 0 end

	local targetChar = plr.Character
	if not targetChar then return 0 end

	local targetHRP = targetChar:FindFirstChild("HumanoidRootPart")
	if not targetHRP then return 0 end

	return math.floor((myHRP.Position - targetHRP.Position).Magnitude)
end

local function RemoveESP(plr)
	local data = espObjects[plr]

	if data then
		for _, obj in ipairs(data) do
			if obj and obj.Parent then
				obj:Destroy()
			end
		end

		espObjects[plr] = nil
	end
end

local function UpdateESP()
	if not espActive then
		return
	end

	for _, plr in ipairs(Players:GetPlayers()) do
		if plr ~= LocalPlayer then
			local char = plr.Character

			if char and char.Parent and char:FindFirstChild("HumanoidRootPart") then
				if not espObjects[plr] then
					CreateESP(plr)
				end

				local data = espObjects[plr]

				if data then
					local distanceLabel = data[4]
					local highlight = data[1]

					local dist = GetDistance(plr)

					distanceLabel.Text = "DISTANCE : [" .. dist .. "M]"

					if dist < 10 then
						distanceLabel.TextColor3 = Color3.fromRGB(255, 0, 0)
						highlight.FillColor = Color3.fromRGB(255, 0, 0)
					elseif dist < 30 then
						distanceLabel.TextColor3 = Color3.fromRGB(255, 255, 0)
						highlight.FillColor = Color3.fromRGB(255, 255, 0)
					else
						distanceLabel.TextColor3 = Color3.fromRGB(0, 255, 0)
						highlight.FillColor = Color3.fromRGB(0, 255, 0)
					end
				end
			else
				RemoveESP(plr)
			end
		end
	end

	for plr in pairs(espObjects) do
		if not plr.Parent or not plr.Character then
			RemoveESP(plr)
		end
	end
end

	Mythic = false,

local function GetTimerValue(text)
	if not text or text == "" then
		return nil
	end

	local minutes, seconds = text:match("(%d+):(%d+)")

	if minutes and seconds then
		return (tonumber(minutes) * 60) + tonumber(seconds)
	end

	return nil
end

local function GetTimerText(obj)
	if not obj then
		return nil
	end

	if obj:IsA("TextLabel") or obj:IsA("TextBox") then
		return obj.Text
	end

	local textLabel = obj:FindFirstChildWhichIsA("TextLabel")

	if textLabel then
		return textLabel.Text
	end

	return nil
end

local function Notify(title, text)
	pcall(function()
		game:GetService("StarterGui"):SetCore("SendNotification", {
			Title = title,
			Text = text,
			Duration = 5
		})
	end)
end

local function CheckTimers()
	local trashEntry = workspace.Map:FindFirstChild("TrashEntry")
	if not trashEntry then return end

	local timerFolder = trashEntry:FindFirstChild("Timer")
	if not timerFolder then return end

	local cosmicObj =
		timerFolder:FindFirstChild("BoardPart")
		and timerFolder.BoardPart:FindFirstChild("SurfaceGui")
		and timerFolder.BoardPart.SurfaceGui:FindFirstChild("Cosmic")

	local mythicObj =
		timerFolder:FindFirstChild("BoardPart")
		and timerFolder.BoardPart:FindFirstChild("SurfaceGui")
		and timerFolder.BoardPart.SurfaceGui:FindFirstChild("Mythic")

	local deepObj =
		timerFolder:FindFirstChild("BoardPart2")
		and timerFolder.BoardPart2:FindFirstChild("SurfaceGui")
		and timerFolder.BoardPart2.SurfaceGui:FindFirstChild("Deep")

	local timers = {
		Cosmic = cosmicObj,
		Mythic = mythicObj,
		Deep = deepObj
	}

	for name, obj in pairs(timers) do
		local text = GetTimerText(obj)

		if text then
			local seconds = GetTimerValue(text)

			if seconds and seconds <= 0 then
				if not lastNotify[name] then
					Notify("🐟 " .. string.upper(name), "Spawner Ready!")
					lastNotify[name] = true
				end
			elseif seconds and seconds > 0 then
				lastNotify[name] = false
			end
		end
	end
end

		return
	end

	local tycoonPath = workspace.Map.Tycoons:FindFirstChild("Tycoon" .. myTycoonNumber)
	if not tycoonPath then
		return
	end

	local myTycoon = tycoonPath:FindFirstChild("Tycoon")
	if not myTycoon then
		return
	end

	local collectZone = myTycoon:FindFirstChild("CollectZone")
	if not collectZone then
		return
	end

	local collectPart = collectZone:FindFirstChild("CollectPart")
	if not collectPart or not collectPart:IsA("BasePart") then
		return
	end

	local character = LocalPlayer.Character
	if not character then
		return
	end

	local hrp = character:FindFirstChild("HumanoidRootPart")
	if not hrp then
		return
	end

	hrp.CFrame = collectPart.CFrame * CFrame.new(0, 10, 0)
end

TeleportTab:CreateToggle({
	Name = "Back To Base",
	CurrentValue = false,
	Callback = function(state)
		if state then
			myTycoonNumber = GetMyTycoonNumber()

			if not myTycoonNumber then
				warn("Base tidak ditemukan")
				return
			end

			if connection then
				connection:Disconnect()
			end

			connection = RunService.Heartbeat:Connect(function()
				TeleportToMyBase()
			end)
		else
			if connection then
				connection:Disconnect()
				connection = nil
			end
		end
	end
})
