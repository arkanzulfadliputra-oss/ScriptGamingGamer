local function PlayLoopSound(SoundId)
	if SoundPlayer and SoundPlayer.Parent then
		SoundPlayer:Stop()
		SoundPlayer:Destroy()
	end

	local HumanoidRootPart = Character and Character:FindFirstChild("HumanoidRootPart")

local function RemoveESP(plr)
	if ESPObjects[plr] then
		for _, obj in pairs(ESPObjects[plr]) do
			if obj and obj.Parent then
				obj:Destroy()
			end
		end
		ESPObjects[plr] = nil
	end
end

local function UpdateAllESP()
	for _, plr in pairs(game.Players:GetPlayers()) do
		if ESPActive and plr ~= Plr then
			if plr.Character and plr.Character.Parent then
				if not ESPObjects[plr] then
					CreateESP(plr)
				end
			else
				RemoveESP(plr)
			end
		else
			RemoveESP(plr)
		end
	end
end

	for obj in pairs(EspSCPObjects) do

		end
	end

local function ServerHop()
	local PlaceId = game.PlaceId
	local Http = game:GetService("HttpService")
	local Teleport = game:GetService("TeleportService")
	local Player = game.Players.LocalPlayer
	
	local success, response = pcall(function()
		return Http:GetAsync("https://games.roblox.com/v1/games/" .. PlaceId .. "/servers/Public?limit=100")
	end)
	
	if not success then return end
	
	local servers = Http:JSONDecode(response).data
	local currentServerId = game.JobId
	
	for _, server in pairs(servers) do
		if server.id ~= currentServerId and server.playing < 26 then
			Teleport:TeleportToPlaceInstance(PlaceId, server.id, Player)
			return
		end
	end
	
	Teleport:Teleport(PlaceId)
end
